```json

  "rule_id": "ZP-WS-01",
  "dimension": "WS",
  "condition": "month_branch_major_element == day_master_element",
  "conclusion": "日主得令（旺）",
  "evidence": "E-ZQ-WS-01",
  "source": "《子平真诠·论得令》",
  "priority": 9,
  "note": ""
}
```
