```python
def _calc_start_age(self, year, month, day, hour, direction):
    """Calculate start age with hour-level precision."""
    if not self._has_sxtwl:
        return 0.0

    import sxtwl
    from datetime import datetime, timedelta
    from .time.jd_converter import jd_to_datetime

    birth_dt = datetime(year, month, day, hour, 0, 0)  # Uses hour

    nearest_jieqi_dt = None
    for i in range(1, 33):
        # Search forward/backward for nearest Jieqi
        if direction == +1:
            test_dt = birth_dt + timedelta(days=i)
        else:
            test_dt = birth_dt - timedelta(days=i)

        day_obj = sxtwl.fromSolar(test_dt.year, test_dt.month, test_dt.day)
        if day_obj.hasJieQi():
            jieqi_jd = day_obj.getJieQiJD()
            nearest_jieqi_dt = jd_to_datetime(jieqi_jd)
            break

    if nearest_jieqi_dt is None:
        return 0.0

    # Calculate precise time difference
    delta = nearest_jieqi_dt - birth_dt
    delta_days = delta.total_seconds() / 86400.0

    # Apply traditional conversion: 3 days = 1 year
    start_age = abs(delta_days) / 3.0

    return start_age
```
